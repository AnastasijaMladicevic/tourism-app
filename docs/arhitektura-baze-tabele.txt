1. gradovi
gradovi(
  id        	SERIAL                  PK, NOT NULL,		
  naziv     	VARCHAR(150)            NOT NULL,
  opis      	TEXT                    NULL,
  geolokacija  	GEOMETRY(Point, 4326)   NOT NULL,
  aktivan  	BOOLEAN                 NOT NULL, DEFAULT TRUE,
  kreirano 	TIMESTAMPTZ           	NOT NULL, DEFAULT NOW()
)
id - identifikator
naziv - ime grada
opis - kratak opis grada
geolokacija - GPS koordinate, postGIS point tip
aktivan - ako je false grad se ne prikazuje turistima
kreirano - vreme dodavanja u bazu

2. uloge
uloge(
  id          	SERIAL        PK, NOT NULL,
  naziv       	VARCHAR(30)   NOT NULL, UNIQUE,

3. korisnici
korisnici(
  id			SERIAL        PK, NOT NULL,
  ime			VARCHAR(100)  NOT NULL,
  prezime		VARCHAR(100)  NOT NULL,
  email			VARCHAR(200)  NOT NULL, UNIQUE,
  lozinka_hash		VARCHAR(500)  NOT NULL,
  id_uloge		INT           NOT NULL, FK -> uloge.id ON DELETE RESTRICT,
  verifikovan		BOOLEAN       NOT NULL, DEFAULT FALSE,
  jezik			VARCHAR(5)    NOT NULL, DEFAULT 'sr', CHECK (jezik IN ('sr', 'en', 'de', 'fr', 'it')),
  verif_token		VARCHAR(200)  NULL,
  verif_token_istek	TIMESTAMPTZ   NULL,
  reset_token		VARCHAR(200)  NULL,
  reset_token_istek	TIMESTAMPTZ   NULL,
  aktivan		BOOLEAN       NOT NULL, DEFAULT TRUE,
  kreirano		TIMESTAMPTZ   NOT NULL, DEFAULT NOW(),
  izmenjeno		TIMESTAMPTZ   NOT NULL, DEFAULT NOW()
)
id - identifikator
ime, prezime - licni podaci
email - jedinstven, koristi se za login
lozinka_hash - uvek se cuva kao hash
id_uloge - strani kljuc na tabelu uloge
verifikovan - false dok se ne uradi verifikacija preko email-a
jezik - za prikaz sadrzaja
verif_token - nasumicni token koji se salje emailom pri registraciji, null nakon verifikacija
verif_token_istek - token vazi 24h, posle toga mora da se trazi novi
reset_token - token za reset lozinke, null kad nije aktivan
reset_token_istek - reset token vazi 1h
aktivan - false za suspendovani nalog, korisnik ne moze da se prijavi ali svi njegovi podaci ostaju u bazi
kreirano - vreme dodavanja u bazu
izmenjeno - vreme poslednje izmene podataka

email: "ana@gmail.com"

5. objekti
objekti(
  id                SERIAL                  PK, NOT NULL,
  naziv             VARCHAR(200)            NOT NULL,
  id_tipa           INT 	            NOT NULL, FK -> tipovi_objekata.id ON DELETE CASCADE,
  opis              TEXT                    NULL,
  adresa            VARCHAR(300)            NULL,
  id_destinacije    INT                     NOT NULL, FK -> destinacije.id ON DELETE CASCADE,
  geolokacija       GEOMETRY(Point, 4326)   NOT NULL,
  telefon           VARCHAR(30)             NULL,
  sajt              VARCHAR(300)            NULL,
  radno_vreme       JSONB                   NULL,
  prosecna_ocena    DECIMAL(3,2)            NOT NULL, DEFAULT 0.00,
                    CHECK (prosecna_ocena BETWEEN 0 AND 5),
  broj_recenzija    INT                     NOT NULL, DEFAULT 0,
                    CHECK (broj_recenzija >= 0),
  aktivan           BOOLEAN                 NOT NULL, DEFAULT TRUE,
  kreirano          TIMESTAMPTZ             NOT NULL, DEFAULT NOW(),
  izmenjeno         TIMESTAMPTZ             NOT NULL, DEFAULT NOW()
)
id - identifikator
naziv - ime objekta
id_tipa - identifikator tipa iz tabele tipovi_objekata
opis - opis objekta
adresa - fizicka adresa
id_destinacije - kojoj destinaciji pripada objekat
geolokacija - GPS koordinate, postGIS point tip
telefon, sajt - kontakt podaci
radno_vreme - JSONB format zbog slozenog rasporeda
prosecna_ocena - azurira triger
broj_recenzija - azurira triger
aktivan - false za suspendovano
kreirano - vreme dodavanja u bazu
izmenjeno - vreme poslednje izmene podataka

7. dogadjaji
dogadjaji(
  id              SERIAL          PK, NOT NULL,
  naziv           VARCHAR(200)    NOT NULL,
  id_tipa         INT    	  NOT NULL, FK -> tipovi_dogadjaja.id ON DELETE CASCADE,
  opis            TEXT            NULL,
  datum_pocetak   TIMESTAMPTZ     NOT NULL,
  datum_kraj      TIMESTAMPTZ     NULL,
                  CHECK (datum_kraj IS NULL OR datum_kraj > datum_pocetak),
  id_objekta      INT             NOT NULL, FK -> objekti.id ON DELETE CASCADE,
  cena            DECIMAL(10,2)   NULL, CHECK (cena >= 0),
  max_posetilaca  INT             NULL, CHECK (max_posetilaca > 0),
  aktivan         BOOLEAN         NOT NULL, DEFAULT TRUE,
  kreirano        TIMESTAMPTZ     NOT NULL, DEFAULT NOW(),
  izmenjeno       TIMESTAMPTZ     NOT NULL, DEFAULT NOW()
)
id - identifikator
naziv - ime dogadjaja
id_tipa - identifikator tipa iz tabele tipovi_dogadjaja
opis - opis dogadjaja
datum_pocetak - kada dogadjaj pocinje
datum_kraj - kada se dogadjaj zavrsava
id_objekta - za koji objekat je vezan
cena - cena ulaznice
max_posetilaca - maksimalan kapacitet
aktivan - false za suspendovano
kreirano - vreme dodavanja u bazu
izmenjeno - vreme poslednje izmene podataka
